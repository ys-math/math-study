\section{双積}

\begin{definition} \label{def: IsBiproduct}
$\mathscr{C}$を零対象を持つ圏とし$A, B \in \Ob(\mathscr{C})$とする.\ 
$A \oplus B \in \Ob(\mathscr{C})$と$\pi_0 \in \Mor_{\mathscr{C}}(A \oplus B, A),\ \pi_1 \in \Mor_{\mathscr{C}}(A \oplus B, B),\ \iota_0 \in \Mor_{\mathscr{C}}(A,A \oplus B),\ \iota_1 \in \Mor_{\mathscr{C}}(B, A \oplus B)$によって定まる組$(A \oplus B, \pi, \iota)$は$(A \oplus B, \pi)$が直積で$(A \oplus B, \iota)$が直和であり次の図式が可換になるとき$A$と$B$の\term{そうせき}{双積}{biproduct}であるという.\

\begin{center}
\begin{tikzpicture}[
  arr/.style={line width=.5pt,-{cm to}},
  lab/.style={font=\scriptsize},
]
\def\R{1.5}   % radius of the circle
\def\g{11}    % angular gap (degrees) left around each object on the circle
 
% objects
\node (T) at (90:\R)  {$A$};
\node (L) at (180:\R) {$B$};
\node (Rt) at (0:\R)  {$A$};
\node (Bt) at (270:\R) {$B$};
\node (C) at (0,0)    {$A \oplus B$};
 
% arrows around the circle
\draw[arr] (180-\g:\R) arc[start angle=180-\g, end angle=90+\g,  radius=\R];
\draw[arr] (180+\g:\R) arc[start angle=180+\g, end angle=270-\g, radius=\R];
\draw[arr] (\g:\R)     arc[start angle=\g,     end angle=90-\g,  radius=\R];
\draw[arr] (-\g:\R)    arc[start angle=-\g,    end angle=-90+\g, radius=\R];
 
\node[lab] at (135:\R+.32) {$0_{B,A}$};
\node[lab] at (225:\R+.32) {$\mathrm{id}_B$};
\node[lab] at (45:\R+.32)  {$\mathrm{id}_A$};
\node[lab] at (-45:\R+.32) {$0_{A,B}$};
 
% arrows through the centre
\draw[arr] (L)  -- node[lab,below] {$\iota_1$} (C);
\draw[arr] (Rt) -- node[lab,above] {$\iota_0$} (C);
\draw[arr] (C)  -- node[lab,left]  {$\pi_0$}   (T);
\draw[arr] (C)  -- node[lab,right] {$\pi_1$}   (Bt);
\end{tikzpicture}
\end{center}
\end{definition}


\begin{definition} \label{def: biproductMap}
$\mathscr{C}$を零対象を持つ局所小圏とし$A_0, A_1, B_0, B_1 \in \Ob(\mathscr{C})$とする.\
$f \colon A_0 \to B_0,\ g \colon A_1 \to B_1$を$\mathscr{C}$の射とする.\
$f \oplus g \coloneq  \begin{pmatrix} f & 0_{A_0, B_1} \\ 0_{A_1,B_0} & g \end{pmatrix}$として$f \oplus g \colon A_0 \oplus B_0 \to A_1 \oplus B_1$を定める.\
\end{definition}
