\section{ファイバー積とファイバー和}

\begin{definition}

\begin{itemize}
	\item $\Ob(\bigstar_2) \coloneq 3$
	\item $\Mor(\bigstar_2) \coloneq \{\id_0,\ \id_1,\ \id_2,\ h_0 \colon 0 \to 2,\ h_1 \colon 1 \to 2\}$
\end{itemize}

\begin{definition}
$\mathscr{C}$を局所小圏とし$A, B, C \in \Ob(\mathscr{C}),\ f \in \Mor_{\mathscr{C}}(A, C),\ g \in \Mor_{\mathscr{C}}(B, C)$とする.\
共変関手$F \colon \bigstar_2 \to \mathscr{C}$を次のように定める.\
\begin{itemize}
	\item $F(0) \coloneq A,\ F(1) \coloneq B,\ F(2) \coloneq C$
	\item $F(h_0) \coloneq f,\ F(h_1) \coloneq g$
\end{itemize}
このとき$F$の極限$(f \pb g, \pi)$を$f$と$g$の\term{ふぁいばーせき}{ファイバー積}{fiber product}もしくは\term{ひきもどし}{引き戻し}{pullback}という.\
$f \pb g$を$A \times_C B$と書くこともある.\

\begin{definition}
$\mathscr{C}$を局所小圏とし$A, B, C \in \Ob(\mathscr{C}),\ f \in \Mor_{\mathscr{C}}(C, A),\ g \in \Mor_{\mathscr{C}}(C, B)$とする.\
共変関手$F \colon \bigstar_2^{\op} \to \mathscr{C}$を次のように定める.\
\begin{itemize}
	\item $F(0) \coloneq A,\ F(1) \coloneq B,\ F(2) \coloneq C$
	\item $F(h_0) \coloneq f,\ F(h_1) \coloneq g$
\end{itemize}
$F$の余極限$(f \po g, \iota)$を$f$と$g$の\term{ふぁいばーわ}{ファイバー和}{fiber sum}もしくは\term{おしだし}{押し出し}{pushout}という.\
$f \po g$を$A \amalg_C B$と書くこともある.\
