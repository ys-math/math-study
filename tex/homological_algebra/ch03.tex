\section{モノ射とエピ射}

\begin{definition} \label{def: IsMonomorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$X \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(X, \dom(f))$に対して次の条件が成り立つとき$f$は\term{ものしゃ}{モノ射}{monomorphism}であるといい$f \colon \dom(f) \hookrightarrow \cod(f)$と書く.
\begin{center}
    $f \circ g = f \circ h \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsEpimorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$Y \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(\cod(f), Y)$に対して次の条件が成り立つとき$f$は\term{えぴしゃ}{エピ射}{epimorphism}であるといい$f \colon \dom(f) \twoheadrightarrow \cod(f)$と書く.
\begin{center}
    $g \circ f = h \circ f \implies g = h$
\end{center}
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$f \in \Mor_{\mathscr{C}}(A, B)$を同型射とすると$f$はモノ射かつエピ射である.\
\end{proposition}

\begin{proof}
$X \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(X, A)$とする.\
$f \circ g = f \circ h$と仮定する.\
$g = f^{-1} \circ f \circ g =  f^{-1} \circ f \circ h = h$であるから$f$はモノ射である.\

$Y \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(B, Y)$とする.\
$g \circ f = h \circ f$と仮定する.\
$g = g \circ f \circ f^{-1} = h \circ f \circ f^{-1} = h$であるから$f$はエピ射である.\
\end{proof}

\begin{definition}
$\mathscr{C}$を圏とし$X,Y \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(X, Y),\ r \in \Mor_{\mathscr{C}}(Y, X)$とする.\
$r \circ s = \id_{X}$であるとき組$(r,s)$を\term{れとらくしょんせくしょんぺあ}{レトラクションセクションペア}{retraction-section pair}とよぶ.\
またこのとき$r$を\term{れとらくしょん}{レトラクション}{retraction}もしくは\term{ぶんれつえぴしゃ}{分裂エピ射}{split epimorphism}と呼び$s$を\term{せくしょん}{セクション}{section}もしくは\term{ぶんれつものしゃ}{分裂モノ射}{split monomorphism}と呼ぶ.
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$X,Y \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(X, Y),\ r \in \Mor(Y, X)$とし$(r,s)$をレトラクションセクションペアとする.\
このとき$r$はエピ射であり$s$はモノ射である.
\end{proposition}

\begin{proof}
$A \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(A, X)$とする.\
$s \circ f = s \circ g$と仮定する.\
$f = \id_X \circ f = r \circ s \circ f = r \circ s \circ g = \id_X \circ g = g$なので$s$はモノ射である.\

$B \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(X, B)$とする.\
$f \circ r = g \circ r$と仮定する.\
$f = f \circ \id_X = f \circ r \circ s = g \circ r \circ s = g \circ \id_X = g$なので$r$はエピ射である.\
\end{proof}
