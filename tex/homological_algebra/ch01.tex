\section{アーベル圏}

\begin{definition} \label{def: IsPreadditive}
局所小圏$\mathscr{C}$は次の条件を満たすとき\term{ぜんかほうけん}{前加法圏}{preadditive category}であるという.
\begin{itemize}
    \item 任意の$A, B \in \Ob(\mathscr{C})$に対して写像$\sigma_{(A,B)} \colon \Mor_{\mathscr{C}}(A, B) \times \Mor_{\mathscr{C}}(A, B) \to \Mor_{\mathscr{C}}(A, B)$が定まり次の条件を満たす.\
    ただし任意の$f, g \in \Mor_{\mathscr{C}}(A, B)$に対して$f + g \coloneq \sigma_{(A,B)}(f,g)$とする.
    \begin{itemize}
    \item ある$0_{A,B}^{+} \in \Mor_{\mathscr{C}}(A,B)$が存在して$(\Mor_{\mathscr{C}}(A,B), \sigma_{(A,B)}, 0_{A,B} )$はアーベル群.
    \item 任意の$A, B, C \in \Ob(\mathscr{C})$と$f,g \in \Mor_{\mathscr{C}}(A, B),\ h \in \Mor_{\mathscr{C}}(B, C)$に対して$h \circ (f + g) = h \circ f + f \circ g$成り立つ.
    \item 任意の$A, B, C \in \Ob(\mathscr{C})$と$f,g \in \Mor_{\mathscr{C}}(B, C),\ h \in \Mor_{\mathscr{C}}(A, B)$に対して$(f + g) \circ h = f \circ h + g \circ h$成り立つ.
    \end{itemize}
\end{itemize}
\end{definition}

\begin{definition} \label{def: zeroMorphism}
$\mathscr{C}$を零対象$Z$を持つ局所小圏とし$A, B \in \Ob({\mathscr{C}})$とする.\
$\iexcl_B \in \Mor_{\mathscr{C}}(Z, B)$と$!_A \in \Mor_{\mathscr{C}}(A, Z)$をそれぞれただ1つ存在する射とする.\
このとき$0_{A,B} \coloneq !_A \circ \iexcl_B$を\term{れいしゃ}{零射}{zero morphism}という.
\end{definition}

\begin{lemma} \label{lem: additive_zero_comp}
$\mathscr{C}$を零対象$Z$を持つ前加法圏とする.\
任意の$A, B, C \in \Ob(\mathscr{C})$と$g \in \Mor_{\mathscr{C}}(B, C),\ h \in \Mor_{\mathscr{C}}(C, A)$に対して次の図式が可換になる.
% https://q.uiver.app/#q=WzAsNixbMCwwLCJCIl0sWzEsMCwiQSJdLFsxLDEsIkMiXSxbMywwLCJCIl0sWzQsMCwiQSJdLFs0LDEsIkMiXSxbMSwwLCIwX3tBLEJ9XnsrfSIsMl0sWzAsMiwiZyIsMl0sWzEsMiwiMF97QSxDfV57K30iXSxbNSwzLCIwX3tDLEJ9XnsrfSJdLFs0LDMsIjBfe0EsIEJ9XnsrfSIsMl0sWzUsNCwiaCIsMl1d
\[\begin{tikzcd}
	B & A && B & A \\
	& C &&& C
	\arrow["g"', from=1-1, to=2-2]
	\arrow["{0_{A,B}^{+}}"', from=1-2, to=1-1]
	\arrow["{0_{A,C}^{+}}", from=1-2, to=2-2]
	\arrow["{0_{A, B}^{+}}"', from=1-5, to=1-4]
	\arrow["{0_{C,B}^{+}}", from=2-5, to=1-4]
	\arrow["h"', from=2-5, to=1-5]
\end{tikzcd}\]
\end{lemma}

\begin{proof}
$g \circ 0_{A,B}^{+} = g \circ (0_{A,B}^{+} + 0_{A,B}^{+}) = g \circ 0_{A,B}^{+} + g \circ 0_{A,B}^{+}$であり両辺$g \circ 0_{A,B}^{+}$を引けば$0_{A,C}^{+} = g \circ 0_{A,B}^{+}$となる.\
同様に$0_{A,B}^{+} \circ h = 0_{C,B}^{+}$となる.
\end{proof}

\begin{proposition} \label{prop: additive_zero_eq_zero_morphism}
$\mathscr{C}$を零対象$Z$を持つ前加法圏とする.\
任意の$A,B \in \Ob(\mathscr{C})$に対して次が成り立つ.
\begin{center}
    $0_{A,B}^{+} = 0_{A,B}$
\end{center}
\end{proposition}

\begin{proof}
$\Mor_{\mathscr{C}}(Z, B)$はシングルトンなので$\iexcl_B = 0_{Z,B}^{+}$である.\
\cref{lem: additive_zero_comp}により$0_{A,B} = !_A \circ 0_{Z,B}^{+} = 0_{A,B}^{+}$となる.
\end{proof}

\begin{definition} \label{def: WalkingPairCategory}
圏$\mathbf{2}$を$\Ob(\mathbf{2}) \coloneq 2,\ \Mor(\mathbf{2}) \coloneq \{\id_0, \id_1\}$として定める.
\end{definition}

\begin{definition} \label{def: pairDiagramFunctor}
$\mathscr{C}$を局所小圏とし$A,B \in \Ob(\mathscr{C})$とする.\
共変関手$D_{A,B} \colon \mathbf{2} \to \mathscr{C}$を次のようにして定める.
\begin{itemize}
    \item $D_{A,B}(0) \coloneq A,\ D_{A,B}(1) \coloneq B$とする.
    \item 任意の$j \in 2$に対して$D_{A,B}(\id_j) \coloneq \id_{D_{A,B}(j)}$とする.
\end{itemize}
\end{definition}

\begin{remark} \label{rem: limit_universal_property}
$\mathscr{J}$を小圏とし$\mathscr{C}$を局所小圏とする.\
$F \colon \mathscr{J} \to \mathscr{C}$を共変関手とする.\
$A \in \Ob(\mathscr{C})$と自然変換$\eta \colon C_A \Rightarrow F$の組$(A,\eta)$を錐という.\
錐$(L, \pi)$が$F$の極限であることは以下の条件と同値である.\
\begin{center}
${^{\forall}(A,\eta)} \in \Cone(A, F),\ {^{\exists!}h \colon A \to L},\ {^{\forall}j \in \Ob(\mathscr{J}), \pi_j \circ h = \eta_j}$
\end{center}
つまり任意の$\mathscr{J}$の射$f \colon X \to Y$に対して次の図式を可換にするただ1つの$\mathscr{C}$の射$h \colon A \to L$が存在する.

\begin{definition} \label{def: directProduct}
$\mathscr{C}$を局所小圏とし$A,B \in \Ob(\mathscr{C})$とする.\
$(L, \pi)$を$D_{A,B}$の極限とする.\
このとき$A \times B \coloneq L$を\term{ちょくせき}{直積}{direct product}といい$\pi_0, \pi_1$を\term{しゃえい}{射影}{projection}という.\
任意の$X \in \Ob(\mathscr{C})$と$f \in \Mor_{\mathscr{C}}(X, A),\ g \in \Mor_{\mathscr{C}}(X, B)$に対して次の図式を可換にするただ1つの射$X \to A \times B$を$\langle f, g \rangle$,\ もしくは$\begin{pmatrix} f \\ g \end{pmatrix}$と書く.
% https://q.uiver.app/#q=WzAsNCxbMCwwLCJYIl0sWzEsMSwiQSBcXHRpbWVzIEIiXSxbMSwyLCJBIl0sWzIsMSwiQiJdLFswLDIsImYiLDIseyJjdXJ2ZSI6MX1dLFswLDMsImciLDAseyJjdXJ2ZSI6LTF9XSxbMCwxLCJcXGxhbmdsZSBmLCBnIFxccmFuZ2xlIiwxXSxbMSwyLCJcXHBpXzAiXSxbMSwzLCJcXHBpXzEiLDJdXQ==
\[\begin{tikzcd}
	X && \\
	& {A \times B} & B \\
	& A
	\arrow["{\langle f, g \rangle}"{description}, from=1-1, to=2-2]
	\arrow["g", curve={height=-6pt}, from=1-1, to=2-3]
	\arrow["f"', curve={height=6pt}, from=1-1, to=3-2]
	\arrow["{\pi_1}"', from=2-2, to=2-3]
	\arrow["{\pi_0}", from=2-2, to=3-2]
\end{tikzcd}\]
\end{definition}

\begin{definition} \label{def: directSum}
$\mathscr{C}$を局所小圏とし$A,B \in \Ob(\mathscr{C})$とする.\
$(Q, \iota)$を$D_{A,B}$の余極限とする.\
このとき$A \amalg B \coloneq Q$を\term{ちょくわ}{直和}{direct sum}といい$\iota_0, \iota_1$を\term{にゅうしゃ}{入射}{injection}という.\
任意の$X \in \Ob(\mathscr{C})$と$f \in \Mor_{\mathscr{C}}(A, X),\ g \in \Mor_{\mathscr{C}}(B, X)$に対して次の図式を可換にするただ1つの射$A \amalg B \to X$を$\lbrack f, g \rbrack$,\ もしくは$\begin{pmatrix} f & g \end{pmatrix}$と書く.
% https://q.uiver.app/#q=WzAsNCxbMCwwLCJYIl0sWzEsMSwiQSBcXGFtYWxnIEIiXSxbMSwyLCJBIl0sWzIsMSwiQiJdLFsyLDAsImYiLDAseyJjdXJ2ZSI6LTF9XSxbMywwLCJnIiwyLHsiY3VydmUiOjF9XSxbMSwwLCJcXGxicmFjayBmLCBnIFxccmJyYWNrIiwxXSxbMiwxLCJcXGlvdGFfMCIsMl0sWzMsMSwiXFxpb3RhXzEiXV0=
\[\begin{tikzcd}
	X && \\
	& {A \amalg B} & B \\
	& A
	\arrow["{\lbrack f, g \rbrack}"{description}, from=2-2, to=1-1]
	\arrow["g"', curve={height=6pt}, from=2-3, to=1-1]
	\arrow["{\iota_1}", from=2-3, to=2-2]
	\arrow["f", curve={height=-6pt}, from=3-2, to=1-1]
	\arrow["{\iota_0}"', from=3-2, to=2-2]
\end{tikzcd}\]
\end{definition}

\begin{proposition} \label{prop: direct_product_and_direct_sum_of_biproduct}
$\mathscr{C}$を前加法圏とし$A, B \in \Ob(\mathscr{C})$とする.\
$(A \oplus B, \pi_0, \pi_1, \iota_0, \iota_1)$が$A$と$B$の双積であるとき$(A \oplus B, \pi)$は\cref{def: pairDiagramFunctor}で定めた$D_{A,B}$の極限であり$(A \oplus B, \iota)$は$D_{A,B}$の余極限である.
\end{proposition}

\begin{proof}
$X \in \Ob(\mathscr{C})$とし$f \in \Mor_{\mathscr{C}}(X, A),\ g \in \Mor_{\mathscr{C}}(X, B)$とする.\
$h \coloneq \iota_0 \circ f + \iota_1 \circ g$とすると
\begin{align*}
\pi_0 \circ h &= \pi_0 \circ \iota_0 \circ f + \pi_0 \circ \iota_1 \circ g \\
	      &= \id_{A} \circ f + 0_{B, A} \circ g \\
	      &= f + 0_{X, A} \\
	      &= f
\end{align*}
となり同様に$\pi_1 \circ h = g$となる.

$\pi_0 \circ h' = f$かつ$\pi_1 \circ h = g$とすると
\begin{align*}
h' &= \id_{A \oplus B} \circ h' \\
   &= (\iota_0 \circ  \pi_0 + \iota_1 \circ \pi_1) \circ h' \\
   &= \iota_0 \circ f + \iota_1 \circ g \\
   &= h
\end{align*}
となるので$(A \oplus B, \pi)$は$D_{A,B}$の極限である.\

$X \in \Ob(\mathscr{C})$とし$f \in \Mor_{\mathscr{C}}(A, X),\ g \in \Mor_{\mathscr{C}}(B, X)$とする.\
$h \coloneq f \circ \pi_0 + g \circ \pi_1$とすると
\begin{align*}
h \circ \iota_0 &= f \circ \pi_0 \circ \iota_0 + g \circ \pi_1 \circ \iota_0 \\
		&= f \circ \id_{A} + g \circ 0_{A, B} \\
		&= f + 0_{A, X} \\
		&= f
\end{align*}
となり同様に$h \circ \iota_1 = g$となる.

$h' \circ \iota_0 = f$かつ$h' \circ \iota_1 = g$とすると
\begin{align*}
h' &= h' \circ \id_{A \oplus B} \\
   &= h' \circ (\iota_0 \circ \pi_0 + \iota_1 \circ \pi_1) \\
   &= f \circ \pi_0 + g \circ \pi_1 \\
   &= h
\end{align*}
となるので$(A \oplus B, \iota)$は$D_{A,B}$の余極限である.
\end{proof}

\begin{proposition} \label{prop: biproduct_of_direct_product}
$\mathscr{C}$を前加法圏とし$A, B \in \Ob(\mathscr{C})$とする.\
$(A \times B, \pi)$を$D_{A,B}$の極限とする.\

\begin{proposition} \label{prop: biproduct_of_direct_sum}
$\mathscr{C}$を前加法圏とし$A, B \in \Ob(\mathscr{C})$とする.\
$(A \amalg B, \iota)$を$D_{A,B}$の余極限とする.\
このとき$\pi_0 \coloneq \lbrack \id_A, 0_{A, B} \rbrack$とし$\pi_1 \coloneq \lbrack 0_{B, A}, \id_B \rbrack$とすると$(A \amalg B, \pi_0, \pi_1, \iota_0, \iota_1)$は双積となる.
\end{proposition}

\begin{proof}
\cref{prop: biproduct_of_direct_product}の双対である.
\end{proof}

\begin{definition} \label{def: IsAdditive}

\end{definition}

\begin{definition} \label{def: cokernel}
	$K_f$の余極限$(Q_{K_f}, \iota_{K_f})$を$f$の\term{よかく}{余核}{cokernel}といい$\Coker(f) \coloneq Q_{K_f},\ c_f \coloneq (\iota_{K_f})_0 \colon \cod(f) \to \Coker(f)$とする.
\end{definition}

\begin{definition} \label{def: IsMonomorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$X \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(X, \dom(f))$に対して次の条件が成り立つとき$f$は\term{ものしゃ}{モノ射}{monomorphism}であるといい$f \colon \dom(f) \hookrightarrow \cod(f)$と書く.
\begin{center}
    $f \circ g = f \circ h \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsEpimorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$Y \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(\cod(f), Y)$に対して次の条件が成り立つとき$f$は\term{えぴしゃ}{エピ射}{epimorphism}であるといい$f \colon \dom(f) \twoheadrightarrow \cod(f)$と書く.
\begin{center}
    $g \circ f = h \circ f \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsAbelian}
加法圏$\mathcal{A}$は次の条件を満たすとき\term{あーべるけん}{アーベル圏}{Abelian category}であるという.
\begin{itemize}
    \item 任意の$f \in \Mor(\mathrm{C})$に対して$f$の核と余核が存在する.
    \item 任意のモノ射$f \in \Mor({\mathscr{C}})$に対してある$g \in \Mor_{\mathscr{C}}$が存在して$f = k_g$となる.
    \item 任意のエピ射$f \in \Mor({\mathscr{C}})$に対してある$g \in \Mor_{\mathscr{C}}$が存在して$f = c_g$となる.
\end{itemize}
\end{definition}

\begin{definition} \label{def: image}
$\mathscr{C}$を核と余核をもつ加法圏とし$f \in \Mor({\mathscr{C}})$とする.\
$\Im(f) \coloneq \Ker(c_f)$とし$i_f \coloneq k_{c_f}$とする.\
組$(\Im(f), i_f)$を$f$の\term{ぞう}{像}{image}と呼ぶ.
\end{definition}
