\section{モノ射とエピ射}

\begin{definition} \label{def: IsMonomorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$X \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(X, \dom(f))$に対して次の条件が成り立つとき$f$は\term{ものしゃ}{モノ射}{monomorphism}であるといい$f \colon \dom(f) \hookrightarrow \cod(f)$と書く.
\begin{center}
    $f \circ g = f \circ h \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsEpimorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$Y \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(\cod(f), Y)$に対して次の条件が成り立つとき$f$は\term{えぴしゃ}{エピ射}{epimorphism}であるといい$f \colon \dom(f) \twoheadrightarrow \cod(f)$と書く.
\begin{center}
    $g \circ f = h \circ f \implies g = h$
\end{center}
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$f \in \Mor_{\mathscr{C}}(A, B)$を同型射とすると$f$はモノ射かつエピ射である.\
\end{proposition}

\begin{proof}
$X \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(X, A)$とする.\
$f \circ g = f \circ h$と仮定する.\
$g = f^{-1} \circ f \circ g =  f^{-1} \circ f \circ h = h$であるから$f$はモノ射である.\

$Y \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(B, Y)$とする.\
$g \circ f = h \circ f$と仮定する.\
$g = g \circ f \circ f^{-1} = h \circ f \circ f^{-1} = h$であるから$f$はエピ射である.\
\end{proof}

\begin{definition}
$\mathscr{C}$を圏とし$A,B \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(A, B),\ r \in \Mor_{\mathscr{C}}(B, A)$とする.\
$r \circ s = \id_{A}$であるとき組$(r,s)$を\term{れとらくしょんせくしょんぺあ}{レトラクションセクションペア}{retraction-section pair}とよぶ.\
またこのとき$r$を\term{れとらくしょん}{レトラクション}{retraction}もしくは\term{ぶんれつえぴしゃ}{分裂エピ射}{split epimorphism}と呼び$s$を\term{せくしょん}{セクション}{section}もしくは\term{ぶんれつものしゃ}{分裂モノ射}{split monomorphism}と呼ぶ.
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$A,B \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(A, B),\ r \in \Mor(B, A)$とし$(r,s)$をレトラクションセクションペアとする.\
このとき$r$はエピ射であり$s$はモノ射である.
\end{proposition}

\begin{proof}
$X \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(X, A)$とする.\
$s \circ f = s \circ g$と仮定する.\
$f = \id_A \circ f = r \circ s \circ f = r \circ s \circ g = \id_A \circ g = g$なので$s$はモノ射である.\

$Y \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(A, Y)$とする.\
$f \circ r = g \circ r$と仮定する.\
$f = f \circ \id_A = f \circ r \circ s = g \circ r \circ s = g \circ \id_A = g$なので$r$はエピ射である.\
\end{proof}

\begin{proposition}
$\mathscr{C}$を圏とし$A,B,C \in \Ob(\mathscr{C})$とする.\
$m \in \Mor(B, C)$をモノ射とし$f \in \Mor(A, C)$とする.\
このとき$m \circ g = f$を満たす$g \in \Mor_{\mathscr{C}}(A,B)$は存在するならば一意的である.\
\end{proposition}

\begin{proof}
$g,g' \in \Mor_{\mathscr{C}}(A, B)$が$m \circ g = f$と$m \circ g' = f$を満たすと仮定する.\
$m \circ g = f = m \circ g'$であり$m$はモノ射なので$g = g'$である.\
\end{proof}

\begin{definition}
$\mathscr{C}$を圏とし$A,B,C \in \Ob(\mathscr{C})$とする.\
$m \in \Mor(B, C)$をモノ射とし$f \in \Mor(A, C)$とする.\
$m \circ g = f$を満たす$g \in \Mor_{\mathscr{C}}(A, B)$が存在するとき$g$を$m \backslash f$と書く.\ つまり次の式が成り立つ.\
\begin{center}
$m \circ m \backslash f = f$
\end{center}
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$A,B,C \in \Ob(\mathscr{C})$とする.\
$e \in \Mor(A, B)$をエピ射とし$f \in \Mor(A, C)$とする.\
このとき$h \circ e = f$を満たす$h \in \Mor_{\mathscr{C}}(B,C)$は存在するならば一意的である.\
\end{proposition}

\begin{proof}
$h,h' \in \Mor_{\mathscr{C}}(B,C)$が$h \circ e = f$と$h' \circ e = f$を満たすと仮定する.\
$h \circ e = f = h' \circ e$であり$e$はエピ射なので$h = h'$である.\
\end{proof}

\begin{definition}
$\mathscr{C}$を圏とし$A,B,C \in \Ob(\mathscr{C})$とする.\
$e \in \Mor(A, B)$をエピ射とし$f \in \Mor(A, C)$とする.\
$h \circ e = f$を満たす$h \in \Mor_{\mathscr{C}}(B, C)$が存在するとき$h$を$f / e$と書く.\ つまり次の式が成り立つ.\
\begin{center}
$f / e \circ e = f$
\end{center}
\end{definition}
