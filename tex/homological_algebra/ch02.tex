\begin{definition}
$\mathscr{C}$を局所小圏とし$A, B, C \in \Ob(\mathscr{C}),\ f \in \Mor_{\mathscr{C}}(A, C),\ g \in \Mor_{\mathscr{C}}(B, C)$とする.\
共変関手$F \colon \bigstar_2 \to \mathscr{C}$を次のように定める.\
\begin{itemize}
	\item $F(0) \coloneq C,\ F(1) \coloneq A,\ F(2) \coloneq B$
	\item $F(h_1) \coloneq f,\ F(h_2) \coloneq g$
\end{itemize}
このとき$F$の極限$(A \times_C B, \pi)$を\term{ふぁいばーせき}{ファイバー積}{fiber product}もしくは\term{ひきもどし}{引き戻し}{pullback}という.\

\end{definition}
