\section{アーベル圏}

\begin{definition} \label{def: WalkingParallelPairCategory}
圏$\mathbf{P}$を$\Ob(\mathbf{P}) \coloneq 2$とし$\Mor{\mathscr{C}} \coloneq \{\id_0, \id_1, u, v\}$として定める.\
ただし$|\Mor_{\mathbf{P}}(0, 1)| = 2$である.\
これを\term{へいこうついのけん}{平行対の圏}{walking parallel pair category}という.\
平行対の圏を$\bullet \rightrightarrows \bullet$と書くこともある.

\begin{definition} \label{def: kernel}
$\mathscr{C}$を零対象を持つ圏とし$f \in \Mor({\mathscr{C}})$とする.\
$\mathbf{P}$を平行対の圏とし共変関手$K_f \colon \mathbf{P} \to \mathscr{C}$を次のようにして定める.
\begin{itemize}
    \item $K_f(0) \coloneq \dom(f),\ K_f(1) \coloneq \cod(f)$とする.
    \item $K_f(u) \coloneq f,\ K_f(v) \coloneq 0_{\dom(f), \cod(f)}$とする.
\end{itemize}
$K_f$の極限$(L_{K_f}, \pi_{K_f})$を$f$の\term{かく}{核}{kernel}といい$\Ker(f) \coloneq L_{K_f},\ \ker(f) \coloneq (\pi_{K_f})_0 \colon \Ker(f) \to \dom(f)$とする.\
つまり任意の$X \in \Ob(\mathscr{C})$と$(X, \eta) \in \Cone(X,K_f)$に対して次の図式を可換にする$g \colon X \to \Ker(f)$がただ1つ存在する.

\begin{remark} \label{rem: kernel_cone_zero_leg}
\cref{def: kernel}の仮定のもと任意の$X \in \Ob(\mathscr{C})$と$(X, \eta) \in \Cone(X,K_f)$に対して$\eta_1 = 0_{\dom(f), \cod(f)} \circ \eta_0 = 0_{X, \cod(f)}$である.
\end{remark}

\begin{definition} \label{def: cokernel}
	$K_f$の余極限$(Q_{K_f}, \iota_{K_f})$を$f$の\term{よかく}{余核}{cokernel}といい$\Coker(f) \coloneq Q_{K_f},\ \coker(f) \coloneq (\iota_{K_f})_0 \colon \cod(f) \to \Coker(f)$とする.
\end{definition}

\begin{definition} \label{def: image}
$\mathscr{C}$を核と余核をもつ加法圏とし$f \in \Mor({\mathscr{C}})$とする.\
$\Im(f) \coloneq \Ker(\coker(f))$とし$\im(f) \coloneq \ker(\coker(f))$とする.\
組$(\Im(f), \im(f))$を$f$の\term{ぞう}{像}{image}と呼ぶ.
\end{definition}

\begin{definition} \label{def: IsMonomorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$X \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(X, \dom(f))$に対して次の条件が成り立つとき$f$は\term{ものしゃ}{モノ射}{monomorphism}であるといい$f \colon \dom(f) \hookrightarrow \cod(f)$と書く.
\begin{center}
    $f \circ g = f \circ h \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsEpimorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$Y \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(\cod(f), Y)$に対して次の条件が成り立つとき$f$は\term{えぴしゃ}{エピ射}{epimorphism}であるといい$f \colon \dom(f) \twoheadrightarrow \cod(f)$と書く.
\begin{center}
    $g \circ f = h \circ f \implies g = h$
\end{center}
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$f \in \Mor_{\mathscr{C}}(A, B)$を同型射とすると$f$はモノ射かつエピ射である.\
\end{proposition}

\begin{proof}
$X \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(X, A)$とする.\
$f \circ g = f \circ h$と仮定する.\
$g = f^{-1} \circ f \circ g =  f^{-1} \circ f \circ h = h$であるから$f$はモノ射である.\

$Y \in \Ob(\mathscr{C})$として$g, h \in \Mor_{\mathscr{C}}(B, Y)$とする.\
$g \circ f = h \circ f$と仮定する.\
$g = g \circ f \circ f^{-1} = h \circ f \circ f^{-1} = h$であるから$f$はエピ射である.\
\end{proof}

\begin{lemma} \label{lem: kernel_and_mono_iff_zero_leg}
$\mathscr{C}$を零対象を持つ圏とする.\
$X,Y \in \Ob(\mathscr{C})$とし$f \in \Mor_{\mathscr{C}}(X,Y)$とすると次が成り立つ.\
\begin{center}
	$f$はモノ射$\iff$$\ker(f) = 0_{\Ker(f),X}$
\end{center}
\end{lemma}

$f \circ \ker(f) = 0_{\Ker(f), Y} = f \circ 0_{\Ker(f),X}$なので$\ker(f) = 0_{\Ker(f),X}$となる.\

$\ker(f) = 0_{\Ker(f),X}$であるとする.\
$A \in \Ob(\mathscr{C}),\ g, h \in \Mor_{\mathscr{C}}(A, X)$とすると$g,h$それぞれに対して次の図式を可換にする$!_g, !_h \in \Mor_{\mathscr{C}}(A,\Ker(f))$が存在する.

\begin{lemma} \label{lem: cokernel_and_epi_iff_zero_leg}
$\mathscr{C}$を零対象を持つ圏とする.\
$X,Y \in \Ob(\mathscr{C})$とし$f \in \Mor_{\mathscr{C}}(X,Y)$とすると次が成り立つ.\
\begin{center}
	$f$はエピ射$\iff$$\coker(f) = 0_{Y,\Coker(f)}$ 
\end{center}
\end{lemma}

\begin{proof}
	\cref{lem: kernel_and_mono_iff_zero_leg}の双対である.
\end{proof}

\begin{proposition}
$\mathcal{A}$をアーベル圏とし$f \in \Mor_{\mathcal{A}}(A,B)$とする.\
$f$がモノ射かつエピ射であるとき$f$は同型射となる.\
\end{proposition}

\begin{definition} \label{def: IsAbelian}
圏$\mathcal{A}$は次の条件を満たすとき\term{あーべるけん}{アーベル圏}{Abelian category}であるという.
\begin{itemize}
    \item 零対象$Z \in \Ob(\mathcal{A})$が存在する.
    \item 任意の$A,B \in \Ob(\mathcal{A})$に対して直積$(A \times B, \pi)$が存在する.
    \item 任意の$f \in \Mor(\mathcal{A})$に対して$f$の核と余核が存在する.
    \item 任意のモノ射$f \in \Mor(\mathscr{C})$に対してある$g \in \Mor(\mathscr{C})$が存在して$f = \ker(g)$となる.
    \item 任意のエピ射$f \in \Mor(\mathscr{C})$に対してある$g \in \Mor(\mathscr{C})$が存在して$f = \coker(g)$となる.
\end{itemize}
\end{definition}

\begin{proof}
$f$はモノ射なのである$C \in \Ob(\mathcal{A})$と$h \in \Mor_{\mathcal{A}}(B, C)$が存在して$f = \ker(h)$となる.\
よって$h \circ f = 0_{A,C} = 0_{B,C} \circ f$であり$f$はエピ射なので$h = 0_{B,C}$である.\
$h \circ \id_B = 0_{B,C}$なので核の普遍性により$f \circ g = \id_B$を満たす$g \colon B \to A$がただ1つ存在する.\
$f \circ g \circ f = \id_B \circ f = f = f \circ \id_A$であり$f$はモノ射なので$g \circ f = \id_A$である.\
よって$f$は同型射である.\
\end{proof}

\begin{remark}
以下断らない限りアーベル圏は局所的に小さいことを仮定する.
\end{remark}

\begin{definition}
$\mathscr{C}$を圏とし$X,Y \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(X, Y),\ r \in \Mor_{\mathscr{C}}(Y, X)$とする.\
$r \circ s = \id_{X}$であるとき組$(r,s)$を\term{れとらくしょんせくしょんぺあ}{レトラクションセクションペア}{retraction-section pair}とよぶ.\
またこのとき$r$を\term{れとらくしょん}{レトラクション}{retraction}もしくは\term{ぶんれつえぴしゃ}{分裂エピ射}{split epimorphism}と呼び$s$を\term{せくしょん}{セクション}{section}もしくは\term{ぶんれつものしゃ}{分裂モノ射}{split monomorphism}と呼ぶ.
\end{definition}

\begin{proposition}
$\mathscr{C}$を圏とし$X,Y \in \Ob(\mathscr{C})$とする.\
$s \in \Mor(X, Y),\ r \in \Mor(Y, X)$とし$(r,s)$をレトラクションセクションペアとする.\
このとき$r$はエピ射であり$s$はモノ射である.
\end{proposition}

\begin{proof}
$A \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(A, X)$とする.\
$s \circ f = s \circ g$と仮定する.\
$f = \id_X \circ f = r \circ s \circ f = r \circ s \circ g = \id_X \circ g = g$なので$s$はモノ射である.\

$B \in \Ob(\mathscr{C})$とし$f,g \in \Mor_{\mathscr{C}}(X, B)$とする.\
$f \circ r = g \circ r$と仮定する.\
$f = f \circ \id_X = f \circ r \circ s = g \circ r \circ s = g \circ \id_X = g$なので$r$はエピ射である.\
\end{proof}

\begin{proposition}
$\mathcal{A}$をアーベル圏とし$A, B \in \Ob(\mathcal{A})$とする.\
$(A \times B, \pi)$を直積とする.
$\iota_0 \coloneq \begin{pmatrix} \id_A \\ 0_{A,B} \end{pmatrix},\ \iota_1 \coloneq \begin{pmatrix} 0_{B,A} \\ \id_B \end{pmatrix}$とすると$(A \times B, \pi, \iota)$は双積になる.
\end{proposition}

\begin{proof}
$\pi_0 \circ \iota_0 = \id_A$なので$(\pi_0, \iota_0)$はレトラクションセクションペアである.\
$X \in \Ob(\mathcal{A})$として$f \in \Mor_{\mathcal{A}}(X, A \times B)$とする.\
$\pi_1 \circ f = 0_{A,B}$であると仮定する.\
$\iota_0 \circ u = f$となる$u \in \Mor_{\mathcal{A}}(X, A)$が存在したとすると$\pi_0 \circ f = u$である.\
そこで$u \coloneq \pi_0 \circ f$とすると$\pi_0 \circ (\iota_0 \circ u) = u = \pi_0 \circ f$であり$\pi_1 \circ (\iota_0 \circ u) = 0_{A,B} \circ u = 0_{A,B} = \pi_1 \circ f$である.\
したがって直積の普遍性により$\iota_0 \circ u = f$である.\
よって次の図式が可換になり$\Ker(\pi_1) = A,\ \ker(\pi_1) = \iota_0$である.\

$\pi_1$は$レトラクション$であるからエピ射である.\
よってある$Y \in \Ob(\mathcal{A})$と$g \in \Mor_{\mathcal{A}}(Y, A \times B)$が存在して$\pi_1 = \coker(g)$となる.\
$X \in \Ob(\mathcal{A})$とし$f \in \Mor_{\mathcal{A}}(A \times B, X)$が$f \circ \iota_0 = 0_{A,X}$を満たすとする.\
$\pi_1 = \coker(g)$なので$\pi_1 \circ g = 0_{Y,B}$である.\
よって核の普遍性により$g = \iota_0 \circ v$となる$v \colon Y \to A$がただ1つ存在する.\
$f \circ \iota_0 = 0_{A,X}$なので$f \circ g = f \circ \iota_0 \circ v = 0_{A,X} \circ v = 0_{Y,X}$となる.\
したがって$\pi_1 = \coker(g)$なので余核の普遍性により$f = \pi_1 \circ u$となる$u \colon B \to X$がただ1つ存在する.\
ゆえに$\pi_1 = \coker(\iota_0)$である.\
同様に$\pi_0 = \coker(\iota_1)$となる.\

\end{proof}

\begin{proposition}
アーベル圏は加法圏である.
\end{proposition}

\begin{proof}

任意の$A,B \in \Ob(\mathcal{A})$に対して$\Mor_{\mathcal{A}}(A,B)$がアーベル群となることを示す.\
$(A \oplus A, \pi^A, \iota^A)$と$(B \oplus B, \pi^B, \iota^B)$をそれぞれ双積とする.\
$f,g \in \Mor_{\mathcal{A}}(A,B)$に対して$f + g \coloneq \begin{pmatrix} \id_B & \id_B \end{pmatrix} \circ (f \oplus g) \circ \begin{pmatrix} \id_A \\ \id_A \end{pmatrix}$とする.\
