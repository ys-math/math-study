\section{アーベル圏}

\begin{definition} \label{def: WalkingParallelPairCategory}
圏$\mathbf{P}$を$\Ob(\mathbf{P}) \coloneq 2$とし$\Mor{\mathscr{C}} \coloneq \{\id_0, \id_1, u, v\}$として定める.\
ただし$|\Mor_{\mathbf{P}}(0, 1)| = 2$である.\
これを\term{へいこうついのけん}{平行対の圏}{walking parallel pair category}という.\
平行対の圏を$\bullet \rightrightarrows \bullet$と書くこともある.

\begin{definition} \label{def: kernel}
$\mathscr{C}$を零対象を持つ圏とし$f \in \Mor({\mathscr{C}})$とする.\
$\mathbf{P}$を平行対の圏とし共変関手$K_f \colon \mathbf{P} \to \mathscr{C}$を次のようにして定める.
\begin{itemize}
    \item $K_f(0) \coloneq \dom(f),\ K_f(1) \coloneq \cod(f)$とする.
    \item $K_f(u) \coloneq f,\ K_f(v) \coloneq 0_{\dom(f), \cod(f)}$とする.
\end{itemize}
$K_f$の極限$(L_{K_f}, \pi_{K_f})$を$f$の\term{かく}{核}{kernel}といい$\Ker(f) \coloneq L_{K_f},\ \ker(f) \coloneq (\pi_{K_f})_0 \colon \Ker(f) \to \dom(f)$とする.\
つまり任意の$X \in \Ob(\mathscr{C})$と$(X, \eta) \in \Cone(X,K_f)$に対して次の図式を可換にする$g \colon X \to \Ker(f)$がただ1つ存在する.

\begin{remark} \label{rem: kernel_cone_zero_leg}
\cref{def: kernel}の仮定のもと任意の$X \in \Ob(\mathscr{C})$と$(X, \eta) \in \Cone(X,K_f)$に対して$\eta_1 = 0_{\dom(f), \cod(f)} \circ \eta_0 = 0_{X, \cod(f)}$である.
\end{remark}

\begin{definition} \label{def: cokernel}
	$K_f$の余極限$(Q_{K_f}, \iota_{K_f})$を$f$の\term{よかく}{余核}{cokernel}といい$\Coker(f) \coloneq Q_{K_f},\ \coker(f) \coloneq (\iota_{K_f})_0 \colon \cod(f) \to \Coker(f)$とする.
\end{definition}

\begin{definition} \label{def: image}
$\mathscr{C}$を核と余核をもつ加法圏とし$f \in \Mor({\mathscr{C}})$とする.\
$\Im(f) \coloneq \Ker(\coker(f))$とし$\im(f) \coloneq \ker(\coker(f))$とする.\
組$(\Im(f), \im(f))$を$f$の\term{ぞう}{像}{image}と呼ぶ.
\end{definition}

\begin{definition} \label{def: IsMonomorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$X \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(X, \dom(f))$に対して次の条件が成り立つとき$f$は\term{ものしゃ}{モノ射}{monomorphism}であるといい$f \colon \dom(f) \hookrightarrow \cod(f)$と書く.
\begin{center}
    $f \circ g = f \circ h \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsEpimorphism}
$\mathscr{C}$を圏とし$f \in \Mor(\mathscr{C})$とする.\
任意の$Y \in \Ob(\mathscr{C})$と$g, h \in \Mor_{\mathscr{C}}(\cod(f), Y)$に対して次の条件が成り立つとき$f$は\term{えぴしゃ}{エピ射}{epimorphism}であるといい$f \colon \dom(f) \twoheadrightarrow \cod(f)$と書く.
\begin{center}
    $g \circ f = h \circ f \implies g = h$
\end{center}
\end{definition}

\begin{definition} \label{def: IsAbelian}
圏$\mathcal{A}$は次の条件を満たすとき\term{あーべるけん}{アーベル圏}{Abelian category}であるという.
\begin{itemize}
    \item 零対象$Z \in \Ob(\mathcal{A})$が存在する.
    \item 任意の$A,B \in \Ob(\mathcal{A})$に対して$D_{A,B} \colon \mathbf{2} \to \mathcal{A}$の極限$(A \times B, \pi)$が存在する.
    \item 任意の$f \in \Mor(\mathcal{A})$に対して$f$の核と余核が存在する.
    \item 任意のモノ射$f \in \Mor(\mathscr{C})$に対してある$g \in \Mor(\mathscr{C})$が存在して$f = \ker(g)$となる.
    \item 任意のエピ射$f \in \Mor(\mathscr{C})$に対してある$g \in \Mor(\mathscr{C})$が存在して$f = \coker(g)$となる.
\end{itemize}
\end{definition}

\begin{proposition}
局所的に小さいアーベル圏は加法圏である.
\end{proposition}

\begin{proof}
$\mathcal{A}$を局所的に小さいアーベル圏とする.\
任意の$A,B \in \Ob(\mathcal{A})$に対して$\Mor_{\mathcal{A}}(A,B)$がアーベル群となることを示す.\
$f,g \in \Mor_{\mathcal{A}}(A,B)$に対して$f + g \coloneq \begin{pmatrix} \id_B & \id_B \end{pmatrix} \circ (f \oplus g) \circ \begin{pmatrix} \id_A \\ \id_A \end{pmatrix}$とする.\ 次の図式が可換になるので$f + 0_{A, B} = f$である.\
\begin{center}
\begin{tikzpicture}[
    commutative diagrams/every diagram,
    x={(1cm,0cm)}, y={(0.7cm,0.5cm)}, z={(0cm,1cm)}
  ]
  % vertices
  \node (BB) at (0,0,4.6)   {$B \oplus B$};
  \node (BL) at (-5,0,0)    {$B$};
  \node (BR) at (5,0,0)     {$B$};
  \node (A0) at (0,-3.6,0)  {$A$};
  \node (A1) at (0,3.6,0)   {$A$};
  \node (AA) at (0,0,0)     {$A \oplus A$};
 
  \path[commutative diagrams/.cd, every arrow, every label]
    % back corner of the base (drawn first: it passes behind the axis)
    (A1) edge node[swap,pos=.75] {$0_{A,B}$} (BL)
    (A1) edge node[swap,pos=.6] {$0_{A,B}$} (BR)
    (A1) edge node[swap,pos=.5] {$\iota_1^A$} (AA)
    % vertical axis
    (AA) edge[commutative diagrams/crossing over]
      node[commutative diagrams/description, pos=.50, font=\footnotesize]
      {$\begin{pmatrix} \begin{pmatrix} f & 0_{A,B} \end{pmatrix} \\
                        \begin{pmatrix} 0_{A,B} & 0_{A,B} \end{pmatrix} \end{pmatrix}$} (BB)
    % product cone
    (BB) edge node[swap] {$\pi_0^B$} (BL)
    (BB) edge node {$\pi_1^B$} (BR)
    % mediating maps out of the centre (labels on the near side of the base)
    (AA) edge node[pos=.55] {$\begin{pmatrix} f & 0_{A,B} \end{pmatrix}$} (BL)
    (AA) edge node[pos=.3] {$\begin{pmatrix} 0_{A,B} & 0_{A,B} \end{pmatrix}$} (BR)
    % front corner of the base (labels on the outside)
    (A0) edge node {$f$} (BL)
    (A0) edge node[swap] {$0_{A,B}$} (BR)
    (A0) edge node[swap,pos=.5] {$\iota_0^A$} (AA);
\end{tikzpicture}
\end{center}
\end{proof}
