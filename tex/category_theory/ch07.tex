\section{コンマ圏}

\begin{definition}
$\mathscr{A},\ \mathscr{B},\ \mathscr{C}$を圏とし$S \colon \mathscr{A} \to \mathscr{C},\ T \colon \mathscr{B} \to \mathscr{C}$を共変関手とする.\
次のようにして\term{こんまけん}{コンマ圏}{comma category}\ $S \downarrow T$が定まる.
\begin{itemize}
	\item 対象を$A \in \Ob(\mathscr{A}),\ B \in \Ob(\mathscr{B}),\ h \in \Mor_{\mathscr{C}}(S(A), T(B))$の組$(A, B, h)$とする.
	\item 射$(A, B, h) \to (A', B', h')$を$f \in \Mor_{\mathscr{A}}(A, A')$と$g \in \Mor_{\mathscr{B}}(B, B')$の組$(f, g)$で次の図式を可換にするものとする.
% https://q.uiver.app/#q=WzAsNCxbMCwwLCJTKEEpIl0sWzEsMCwiVChCKSJdLFswLDEsIlMoQScpIl0sWzEsMSwiVChCJykiXSxbMCwxLCJoIl0sWzEsMywiVChnKSJdLFswLDIsIlMoZikiLDJdLFsyLDMsImgnIl1d
\[\begin{tikzcd}
	{S(A)} & {T(B)} \\
	{S(A')} & {T(B')}
	\arrow["h", from=1-1, to=1-2]
	\arrow["{S(f)}"', from=1-1, to=2-1]
	\arrow["{T(g)}", from=1-2, to=2-2]
	\arrow["{h'}", from=2-1, to=2-2]
\end{tikzcd}\]
	\begin{itemize}
		\item 射$(f,g)$と$(f',g')$の合成は$(f,g) \circ (f',g') \coloneq (f' \circ f, g' \circ g)$で定める.
		\item 恒等射は$\id_{(A,B,h)} \coloneq (\id_{A}, \id_{B})$で定める.
	\end{itemize}
\end{itemize}
\end{definition}

\begin{definition}
$\mathscr{A},\ \mathscr{B},\ \mathscr{C}$を圏とし$S \colon \mathscr{A} \to \mathscr{C},\ T \colon \mathscr{B} \to \mathscr{C}$を共変関手とする.\
\term{しゃえいかんしゅ}{射影関手}{projection functor}\ $P \colon (S \downarrow T) \to \mathscr{A}),\ Q \colon (S \downarrow T) \to \mathscr{B}$を次のようにして定める.
\begin{center}
$P(A,B,h) \coloneq A,\quad P(f,g) \coloneq f,\quad Q(A,B,h) \coloneq B,\quad Q(f,g) \coloneq g$
\end{center}
\end{definition}

\begin{proposition}
$\mathscr{A},\ \mathscr{B},\ \mathscr{C}$を圏とし$S \colon \mathscr{A} \to \mathscr{C},\ T \colon \mathscr{B} \to \mathscr{C}$を共変関手とする.\
$(A,B,h) \in \Ob(S \downarrow T)$に対して$\theta_{(A,B,h)} \coloneq h$とすると$\theta \colon SP \Rightarrow TQ$は自然変換になる.
\end{proposition}

\begin{proof}
$(f,g) \in \Mor_{S \downarrow T}((A,B,h), (A',B',h'))$に対して$\theta_{(A',B',h')} \circ SP(f,g) = h' \circ S(f)$となるがコンマ圏の定義より$h' \circ S(f) = T(g) \circ h$であり,\
$T(g) \circ h = TQ(f,g) \circ \theta_{(A,B,h)}$なので次の図式が可換になる.
% https://q.uiver.app/#q=WzAsNCxbMCwwLCJTUChBLEIsaCkiXSxbMSwwLCJUUShBLEIsaCkiXSxbMCwxLCJTUChBJyxCJyxoJykiXSxbMSwxLCJUUShBJyxCJyxoJykiXSxbMCwxLCJcXHRoZXRhX3soQSxCLGgpfSJdLFsxLDMsIlRRKGYsZykiXSxbMCwyLCJTUChmLGcpIiwyXSxbMiwzLCJcXHRoZXRhX3soQScsQicsaCcpfSJdXQ==
\[\begin{tikzcd}
	{SP(A,B,h)} & {TQ(A,B,h)} \\
	{SP(A',B',h')} & {TQ(A',B',h')}
	\arrow["{\theta_{(A,B,h)}}", from=1-1, to=1-2]
	\arrow["{SP(f,g)}"', from=1-1, to=2-1]
	\arrow["{TQ(f,g)}", from=1-2, to=2-2]
	\arrow["{\theta_{(A',B',h')}}", from=2-1, to=2-2]
\end{tikzcd}\]
したがって$\theta \colon SP \Rightarrow  TQ$は自然変換である.
\end{proof}

\begin{proposition}
	$\mathscr{A},\ \mathscr{B},\ \mathscr{C},\ \mathscr{X}$を圏とし$S \colon \mathscr{A} \to \mathscr{C},\ T \colon \mathscr{B} \to \mathscr{C},\ P' \colon \mathscr{X} \to \mathscr{A},\ Q' \colon \mathscr{X} \to \mathscr{B}$を共変関手とする.\
$\theta' \colon SP' \Rightarrow TQ'$を自然変換とする.\
このとき次の図式を可換し$\theta' = \theta R$を満たす共変関手$R \colon \mathscr{X} \to (S \downarrow T)$がただ1つ存在する.

\end{proposition}
