\section{圏同値}

\begin{definition} \label{def: Equivalence}
$\mathscr{C},\ \mathscr{D}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{D} \to \mathscr{C}$を共変関手とする.\
自然同型$\eta \colon \id_{\mathscr{C}} \Rightarrow GF,\ \varepsilon \colon FG \Rightarrow \id_{\mathscr{D}}$が存在するとき$\mathscr{C}$と$\mathscr{D}$は\term{けんどうち}{圏同値}{equivalence of categories}であるといい組$(F,G,\eta,\varepsilon)$を$\mathscr{C}$と$\mathscr{D}$の間の圏同値と呼ぶ.\ $\mathscr{C}$と$\mathscr{D}$が圏同値であるとき$\mathscr{C} \simeq \mathscr{D}$と書く.
\end{definition}

\begin{definition} \label{def: IsAdjointEquivalence}
$\mathscr{C},\ \mathscr{D}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{D} \to \mathscr{C}$を共変関手とする.\
随伴$F \dashv G$が存在し$F \dashv G \eqcolon (\eta, \varepsilon)$としたとき$\eta$と$\varepsilon$が自然同型なら随伴$F \dashv G$を\term{ずいはんどうち}{随伴同値}{adjoint equivalence}であるという.
\end{definition}

\begin{definition}
$\mathscr{C},\ \mathscr{D}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{C} \to \mathscr{D}$を共変関手とする.\
$\alpha \colon F \Rightarrow G$を自然同型とする.\
このとき自然同型$\alpha^{-1} \colon G \Rightarrow F$を$X \in \Ob(\mathscr{C})$に対して$(\alpha^{-1})_X \coloneq (\alpha_X)^{-1}$で定める.
\end{definition}

\begin{lemma}
$\mathscr{C},\ \mathscr{D},\ \mathscr{E}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{C} \to \mathscr{D}, H \colon \mathscr{D} \to \mathscr{E}$を共変関手とする.\
$\alpha \colon F \Rightarrow G$を自然同型とする.\
このとき$(H \alpha)^{-1} = H(\alpha^{-1})$である.\ 右ヒゲ結合に関しても同様.
\end{lemma}

\begin{proof}
任意の$X \in \Ob(\mathscr{C})$に対して$((H\alpha)^{-1})_X = ((H \alpha)_X)^{-1} = (H(\alpha_X))^{-1} = H((\alpha_X)^{-1}) = H((\alpha^{-1})_X) = (H(\alpha^{-1})_X$である.
\end{proof}

\begin{lemma}
$\mathscr{C},\ \mathscr{D}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{D} \to \mathscr{C}$を共変関手とする.\
$(F,G,\eta,\varepsilon)$を圏同値とする.\
任意の$A \in \Ob(\mathscr{C}),\ B \in \Ob(\mathscr{D})$に対して$GF(\eta_A) = \eta_{GF(A)}$かつ$FG(\varepsilon_B) = \varepsilon_{FG(b)}$が成り立つ.
\end{lemma}

\begin{proof}
$\eta \colon \id_{\mathscr{C}} \Rightarrow GF$は自然変換なので任意の$A \in \Ob(\mathscr{C})$に対して次の図式が可換になる.

\begin{proposition}
$\mathscr{C},\ \mathscr{D}$を圏とし$F \colon \mathscr{C} \to \mathscr{D},\ G \colon \mathscr{D} \to \mathscr{C}$を共変関手とする.\
$(F,G,\eta,\varepsilon)$を圏同値とする.\
$\varepsilon' \coloneq \varepsilon F(\eta^{-1}G) \circ \varepsilon^{-1}(FG)$とすると$F \dashv G \coloneq (\eta, \varepsilon')$は随伴同値である.
\end{proposition}

\begin{proof}
$\varepsilon'$は自然同型の合成から成るので自然同型である.\

$\varepsilon' F \circ F \eta = \id_F$を示す.\
$A \in \Ob(\mathscr{C})$とする.\ 
\end{proof}
