\section{群}
\begin{definition}
集合$M$と写像$\cdot_M \colon M \times M \to M$と$1_M \in M$が次の条件を満たすとき組$(M, \cdot_M, 1_M)$は\textbf{モノイド}であるという.
\def\labelenumi{(\theenumi)}
\begin{enumerate}
    \item $^{\forall}x,y,z \in M,\ (x \cdot_M y) \cdot_M z = x \cdot_M (y \cdot_M z)$
    \item $^{\forall}x \in M,\ 1_M \cdot_M x = x$
\end{enumerate}
更に$^{\forall}x,y \in M,\ x \cdot_M y = y \cdot_M x$も成り立つとき$(M, \cdot_M, 1_M)$は\textbf{可換モノイド}であるという.
$\cdot_M$を\textbf{積},\ $1_M$を\textbf{単位元}と呼ぶ.
\hfill $\square$
\end{definition}

\begin{definition}
$(M, \cdot_M, 1_M)$をモノイドとし$x \in M$とする.\ $x \cdot_M y = 1_M$となる$y \in M$が存在するとき$x$は\textbf{可逆}であるといい,\ このとき$y$を$x$の\textbf{逆元}と呼び$x^{-1_M}$で表す.
\hfill $\square$
\end{definition}

\begin{definition}
$(G, \cdot_G, 1_G)$をモノイドとする.\ 任意の$G$の元が可逆であるとき$(G, \cdot_G, 1_G)$は\textbf{群}であるという.\ $(G, \cdot_G, 1_G)$が可換モノイドのときは\textbf{アーベル群}であるという
\end{definition}

以下モノイド$(M, \cdot_M, 1_M)$や群$(G, \cdot_G, 1_M)$などを単に$M$や$G$と書くこともある.\ 可換モノイドやアーベル群の単位元は$0$と表すことがある.

\begin{definition}
$G, H$を群とし$f \colon G \to H$を写像とする.\ $^{\forall}g_1, g_2 \in G, f(g_1 \cdot_G g_2) = f(g_1) \cdot_H f(g_2)$が成り立つとき$f$は\textbf{群準同型}であるという.\ $f$が可逆なら\textbf{群同型}であるといい群同型$f \colon G \to H$が存在するとき$G$と$H$は\textbf{同型}であるといい$G \cong H$と書く.
\end{definition}
