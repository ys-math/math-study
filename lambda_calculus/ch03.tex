\section{純粋型システム}
\begin{definition}
$\mathcal{S}$を集合,\ $\mathcal{A} \subset \mathcal{S} \times \mathcal{S}$, $\mathcal{R} \subset \mathcal{S} \times \mathcal{S} \times \mathcal{S}$とする.
$S = (\mathcal{S}, \mathcal{A}, \mathcal{R})$を\textbf{仕様}(specification)とよび$\mathcal{S}$, $\mathcal{A}$, $\mathcal{R}$それぞれの元を\textbf{ソート}(sort), \textbf{公理}(axiom), \textbf{規則}(rule)と呼ぶ.
\hfill $\square$
\end{definition}
